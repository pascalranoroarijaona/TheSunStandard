import urllib.request, json, numpy as np
from scipy import stats

def audit_real_bitcoin_rhythm(n_blocks=2016):
    tip = int(urllib.request.urlopen("https://mempool.space/api/blocks/tip/height").read().decode())
    blocks, h = [], tip
    while len(blocks) < n_blocks:
        batch = json.loads(urllib.request.urlopen(f"https://mempool.space/api/blocks/{h}").read().decode())
        blocks.extend(batch); h = batch[-1]["height"] - 1
    ts = np.array([b["timestamp"] for b in blocks[:n_blocks][::-1]], dtype=float)
    dt = np.diff(ts); dt_pos = dt[dt > 0]
    pace = 86400.0 / np.mean(dt)
    ks_d, ks_p = stats.kstest(dt_pos, "expon", args=(0, np.mean(dt_pos)))
    print(f"Observed Pace: {pace:.2f} blocks/day | Target: 144.00 | KS D={ks_d:.4f} (p={ks_p:.3e})")
